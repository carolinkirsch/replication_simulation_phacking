Estimate the empirical Type I error rate of Yuen’s t-test across four trimming parameters, c ∈ {0.1, 0.15, 0.2, 0.25}, and a wide range of sample sizes (up to roughly n = 500 per group), comparing two implementations: DescTools::YuenTTest() and WRS2::yuen(). The two packages compute the winsorized variance differently. DescTools interpolates between order statistics via quantile(type = 7), while WRS2 uses a plain floor-based index cutoff (WRS2:::winvar). They only agree exactly when the interpolation index happens to be a whole number. Sample sizes are chosen in triplets (n, n+1, n+2 around anchors that are multiples of 20) so that all three possible cases — exact agreement, and interpolation in each direction — occur, simultaneously, for all four trim parameters.

functions

data analysis: Yuen’s t-test, two implementations

analyse_data_yuen_methods <- function(n, trim) {
  data <- generate_data_ttest(n_control = n)
  tibble(
    method = c("DescTools", "WRS2"),
    p_value = c(
      YuenTTest(score ~ condition, data, alternative = "two.sided", trim = trim)$p.value,
      yuen(score ~ condition, data, tr = trim)$p.value
    )
  )
}

classify a (n, trim) cell by how DescTools’ and WRS2’s winsorization cutoffs relate

classify_cell <- function(n, trim, tol = 1e-8) {
  x    <- seq_len(n)
  q_lo <- as.numeric(quantile(x, probs = trim, type = 7))   # DescTools cutoff
  o_lo <- x[floor(trim * n) + 1]                             # WRS2 cutoff
  case_when(
    abs(q_lo - o_lo) < tol ~ "A_exact",
    q_lo > o_lo             ~ "B_interp_above",
    TRUE                    ~ "B_interp_below"
  )
}

simulation

design: sample-size triplets shared across trim parameters

c*n is a whole number only if n is a multiple of the period q of c (e.g. q = 10 for c = 0.1, q = 4 for c = 0.25). Using q = 20, the least common multiple of the periods of all four trims (10, 20, 5, 4), makes the interpolation index a whole number for all four trims simultaneously whenever n is a multiple of 20. Around each such anchor, n, n + 1, and n + 2 are simulated to hit all three cases. Anchor points are placed with increasing spacing from n = 20 to n = 500 (capped so triplets don’t exceed 500).

trims <- c(0.10, 0.15, 0.20, 0.25)

# session info is recorded only when the simulation actually runs (not on re-renders from cache)
if (!file.exists(sim_yuen_trim_file)) {
  set.seed(9182)
  plan(multisession, workers = future::availableCores())

  sim_yuen_trim <- par_grid_yuen_trim |>
    mutate(
      analysis_result = future_map2(
        n, trim, analyse_data_yuen_methods,
        .options = furrr_options(seed = TRUE),
        .progress = TRUE
      ),
      case            = map2_chr(n, trim, classify_cell)
    ) |>
    unnest(analysis_result)

  saveRDS(sim_yuen_trim, sim_yuen_trim_file)
  writeLines(capture.output(sessionInfo()), yuen_sessionInfo_file)
} else {
  sim_yuen_trim <- readRDS(sim_yuen_trim_file)
}

results

summary: empirical Type I error by trim, n, case, and method

sim_summary_yuen_trim_file <- paste0("results_UBELIX/sim_yuen_trim_summary_", n_iterations, "iterations.rds")

if (!file.exists(sim_summary_yuen_trim_file)) {
  sim_summary_yuen_trim <- sim_yuen_trim |>
    group_by(trim, n, offset, case, method) |>
    summarize(
      empirical_detection_rate = mean(p_value < alpha),
      empirical_detection_rate_mcse = sqrt(empirical_detection_rate * (1 - empirical_detection_rate) / n()),
      .groups = "drop"
    ) |>
    mutate(
      empirical_detection_rate_95CI_low = empirical_detection_rate - empirical_detection_rate_mcse * 1.96,
      empirical_detection_rate_95CI_upp = empirical_detection_rate + empirical_detection_rate_mcse * 1.96,
    )

  saveRDS(sim_summary_yuen_trim, sim_summary_yuen_trim_file)
} else {
  sim_summary_yuen_trim <- readRDS(sim_summary_yuen_trim_file)
}
